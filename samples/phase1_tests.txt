// Test Case 1: Standard Valid Mini Java Program (PDF Page 4 Example)
// Covers: class, public, static, void, main, String, System.out.println,
//         identifiers, integers, operators, and delimiters.
// -----------------------------------------------------------------------------
class Factorial {
    public static void main(String[] a) {
        System.out.println(new Fac().ComputeFac(10));
    }
}

class Fac {
    public int ComputeFac(int num) {
        int num_aux;
        if (num < 1)
            num_aux = 1;
        else
            num_aux = num * (this.ComputeFac(num - 1));
        return num_aux;
    }
}

// Test Case 2: Arithmetic, Logical, and Relational Operators
// Covers: +, -, *, /, %, ==, !=, <, <=, >, >=, &&, ||, !, =, floats, booleans
// -----------------------------------------------------------------------------
class ExpressionsTest {
    public static void main(String[] args) {
        int a;
        float b;
        boolean flag;

        a = 10 + 5 * 2 - 8 / 4 % 3;
        b = 3.14159 * 2.0;
        flag = (a >= 10) && (b != 0.0) || !(a < 5);

        if (flag == true) {
            System.out.println(a);
        } else {
            System.out.println(0);
        }
    }
}
